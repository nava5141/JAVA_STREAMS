import java.util.*;
import java.util.stream.*;
public class Main
{
	public static void main(String[] args) {
	List<Integer> numbers = Arrays.asList(1,2,3,4,5,6);
	List<Integer> skippedList = numbers.stream()
	                                    .skip(2)
	                                    .collect(Collectors.toList())	; 
	                                    System.out.println(skippedList);
	}
}
